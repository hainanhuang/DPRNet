%% Reuse shared figure/table files that contain their own single-column float
%% wrapper while promoting selected items to a proper two-column float.
\newcommand{\inputcagwidefigure}[1]{%
  \begin{figure*}[!t]
    \begingroup
    \renewenvironment{figure}[1][]{}{}
    \input{#1}
    \endgroup
  \end{figure*}%
}
\newcommand{\inputcagwidetable}[1]{%
  \begin{table*}[!t]
    \begingroup
    \renewenvironment{table}[1][]{}{}
    \input{#1}
    \endgroup
  \end{table*}%
}
